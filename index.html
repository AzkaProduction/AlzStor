Konfigurasi Auto-Topup API (Digiflazz + Midtrans/Tripay)

Atur kredensial API kamu atau salin kode backend Node.js siap pakai agar transaksi otomatis 24 jam.

Tersimpan!
Kode Backend Webhook Otomatis (server.js - Node.js/Express) 
const express = require('express');
const crypto = require('crypto');
const axios = require('axios');
const app = express();
app.use(express.json());

// Webhook otomatis saat pembayaran QRIS berhasil (SETTLEMENT)
app.post('/webhook/payment', async (req, res) => {
  const { order_id, transaction_status, buyer_sku_code, customer_no } = req.body;
  if (transaction_status === 'settlement') {
    const username = process.env.DIGI_USERNAME;
    const apiKey = process.env.DIGI_API_KEY;
    const sign = crypto.createHash('md5').update(username + apiKey + order_id).digest('hex');

    const topupRes = await axios.post('https://api.digiflazz.com/v1/transaction', {
      username,
      buyer_sku_code,
      customer_no,
      ref_id: order_id,
      sign
    });
    console.log('Topup Otomatis Sukses:', topupRes.data);
  }
  res.sendStatus(200);
});

app.listen(3000, () => console.log('Alz Store Auto-Topup Server running on port 3000'));
